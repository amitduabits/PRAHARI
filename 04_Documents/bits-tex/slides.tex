\begin{frame}{Integrator laws in the capture client}
\begin{itemize}\itemsep0.45em
\item Set \code{OPENCV\_FFMPEG\_CAPTURE\_OPTIONS=rtsp\_transport;tcp} before \code{import cv2}
\item Event time is PTS (\code{CAP\_PROP\_POS\_MSEC}), never arrival time, never declared FPS
\item Reconnect backoff starts at 2\,s and caps at 30\,s
\item Decoder warnings at join are logged, not fatal
\item PTS rewind or gap $>$ 5\,s is a scene cut (feeds loop)
\item Consume only. Do not publish. Do not wget \code{/stream/<id>}
\item Tests: \code{tests/test\_integrator\_laws.py}
\end{itemize}
\end{frame}

\begin{frame}{Eighty thousand cameras (DESIGN TARGET)}
\begin{itemize}\itemsep0.6em
\item About 45{,}000 public-domain cameras run analytics at 1\,fps, 720p crop $\approx$ 80\,KB
\item Naive centre: $45{,}000 \times 80\,\mathrm{KB/s} \approx 3.6\,\mathrm{GB/s}$
\item Five regions (Ahmedabad, Surat, Rajkot, Vadodara, Bhuj): $\approx$ 720\,MB/s each
\item SOC wall pulls 16--64 HLS tiles. Everything else is metadata + crops
\item These numbers are design targets, not measured laptop throughput
\end{itemize}
\end{frame}

\begin{frame}{Security in the PoC}
\begin{itemize}\itemsep0.55em
\item Roles: \code{superadmin}, \code{soc\_operator}, \code{dept\_viewer}, \code{auditor}
\item \code{home.viewer} sees Home cameras only; private-permitted mall is hidden
\item HLS/file tiles use a 60\,s HMAC token. Raw RTSP never reaches the browser
\item Sentinel cookie stays on the server. The browser never sees it
\item Private-permitted onboard requires \code{consent=true}
\item Audit: onboard, watchlist edit, ack, report download, operator confirm
\end{itemize}
\end{frame}
