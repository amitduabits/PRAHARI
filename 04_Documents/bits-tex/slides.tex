\begin{frame}{The school-notebook analogy}
\begin{itemize}\itemsep0.65em
\item 26 schools. Each bought a different cupboard for tapes
\item Some cupboards keep 7 days. Some keep 15
\item Head office has a notebook of stolen bicycles and missing children
\item Nobody in the camera room opens that notebook
\item That is Gujarat CCTV: 26 departments, analog and IP, 1{,}000\,km
\item Watchlists exist (VAHAN, SARTHI, eGujCop, AFIS, NAFIS) and sit unused
\end{itemize}
\end{frame}

\begin{frame}{What the State already has}
\begin{itemize}\itemsep0.65em
\item Independent CCTV estates. No statewide camera census
\item Cloud NVR in some estates, local storage in others
\item Sites: border districts, Valsad, Dahod, Somnath, Jamnagar, Dwarka
\item The official goal: use existing infrastructure first
\item A new cupboard for every tape is the wrong first move
\end{itemize}
\end{frame}

\begin{frame}{Integrator laws in the capture client}
\begin{itemize}\itemsep0.45em
\item Set \code{OPENCV\_FFMPEG\_CAPTURE\_OPTIONS=rtsp\_transport;tcp} before \code{import cv2}
\item Event time is PTS (\code{CAP\_PROP\_POS\_MSEC}), never arrival time, never declared FPS
\item Reconnect backoff starts at 2\,s and caps at 30\,s
\item Decoder warnings at join are logged, not fatal
\item PTS rewind or gap $>$ 5\,s is a scene cut (feeds loop)
\item Consume only. Do not wget \code{/stream/<id>}
\item Tests: \code{tests/test\_integrator\_laws.py}
\end{itemize}
\end{frame}

\begin{frame}{Why those laws (ten-year-old version)}
\begin{itemize}\itemsep0.55em
\item UDP across a firewall arrives as a broken picture that looks like a bad model
\item After join, extra frames may arrive fast. Wall-clock time invents flying cars
\item Feeds restart. Reconnecting in a tight loop is shouting at a locked door
\item The tape loops. At the splice, forget the old object ids
\item A downloaded \code{/stream} file is a fake dataset
\end{itemize}
\end{frame}

\begin{frame}{ANPR: options}
\begin{itemize}\itemsep0.55em
\item Train a detector from scratch: no labelled Gujarat set in seven days
\item Require GPU YOLO now: GPU count MEASURED 0 on this laptop
\item Morphology + Tesseract behind \code{recognize()}: CPU, swappable
\item Operator confirm: a readable plate cannot fail a live demo
\item Confirm writes \code{source=operator\_confirm}. Never labelled ANPR
\end{itemize}
\end{frame}

\begin{frame}{ANPR: mechanism}
\begin{itemize}\itemsep0.55em
\item Sampled frame $\to$ plate box $\to$ Tesseract $\to$ Indian normaliser
\item Regex after stripping space and hyphen: \code{\^{}[A-Z]\{2\}[0-9]\{1,2\}[A-Z]\{1,3\}[0-9]\{3,4\}\$}
\item Interface: \code{recognize(frame) -> \{plate, plate\_raw, confidence, crop\}}
\item Missing Tesseract binary: empty plate, no crash
\item MEASURED 03 Sep: confirm on \code{cam04}, confidence 1.0
\item Production swap (same function): YOLO plate detector + PaddleOCR
\end{itemize}
\end{frame}

\begin{frame}{Objects: options}
\begin{itemize}\itemsep0.55em
\item YOLOv8n as the only path: extra dependency, tests would skip
\item OpenCV DNN with large weights: allowed later, not required for pytest
\item Skin-tone blob fallback: a rectangle $\ge 8\%$ of the frame is a person
\item Skipping objects: eval area 05 names person and vehicle detection
\item Classes kept: person, car, motorcycle, bus, truck, bicycle
\end{itemize}
\end{frame}

\begin{frame}{Intrusion on a godown}
\begin{itemize}\itemsep0.55em
\item Food and Civil Supplies watches a warehouse, not a highway
\item ROI lives in \code{cameras.extra\_json}
\item Person box overlapping the ROI by $\ge 30\%$ of its area fires
\item Event: \code{entity\_type=intrusion}, CRITICAL
\item Same camera within 120\,s increments a counter
\item No fourth neural net. Intrusion wraps object boxes
\end{itemize}
\end{frame}

\begin{frame}{Eighty thousand cameras (DESIGN TARGET)}
\begin{itemize}\itemsep0.6em
\item About 45{,}000 public-domain cameras run analytics at 1\,fps, 720p crop $\approx$ 80\,KB
\item Naive centre: $45{,}000 \times 80\,\mathrm{KB/s} \approx 3.6\,\mathrm{GB/s}$
\item Five regions (Ahmedabad, Surat, Rajkot, Vadodara, Bhuj): $\approx$ 720\,MB/s each
\item SOC wall pulls 16--64 HLS tiles. Everything else is metadata + crops
\item These numbers are design targets, not measured laptop throughput
\end{itemize}
\end{frame}
